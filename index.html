            Catatan setup: Akun Admin/Guru disimpan di tabel users. Jika memakai Supabase, buat tabel berikut terlebih dahulu di SQL Editor (dan tambahkan kolom tgl_lahir di tabel siswa): 
            create table users (
  id text primary key,
  username text unique not null,
  password text not null,
  role text not null check (role in ('admin','guru')),
  nama text
);

alter table siswa add column if not exists tgl_lahir date;
